64static uint32_t linux_millis(void)
65{
66 struct timespec ts;
67
68 if (clock_gettime(CLOCK_MONOTONIC, &ts) != 0) {
69 return 0u;
70 }
71
72 return (uint32_t)((uint64_t)ts.tv_sec * 1000u + (uint64_t)ts.tv_nsec / 1000000u);
73}

93static int linux_serial_configure(int fd, unsigned baud)
94{
95 struct termios tio;
96 speed_t speed = linux_baud_to_speed(baud);
97
98 if (tcgetattr(fd, &tio) != 0) {
99 return -1;
100 }
101
102 cfmakeraw(&tio);
103 tio.c_cflag |= (CLOCAL | CREAD);
104 tio.c_cflag &= (tcflag_t)~CSTOPB;
105 tio.c_cflag &= (tcflag_t)~PARENB;
106 tio.c_cflag &= (tcflag_t)~CSIZE;
107 tio.c_cflag |= CS8;
108 tio.c_cc[VMIN] = 0;
109 tio.c_cc[VTIME] = 0;
110
111 if (cfsetispeed(&tio, speed) != 0 || cfsetospeed(&tio, speed) != 0) {
112 return -1;
113 }
114 if (tcsetattr(fd, TCSANOW, &tio) != 0) {
115 return -1;
116 }
117
118 return 0;
119}

121static int linux_uartbin_app_write(const uint8_t *data, size_t len, void *user)
122{
123 linux_uartbin_app_t *port = (linux_uartbin_app_t *)user;
124 size_t sent = 0u;
125
126 while (sent < len) {
127 ssize_t n = write(port->fd, &data[sent], len - sent);
128 if (n > 0) {
129 sent += (size_t)n;
130 continue;
131 }
132 if (n < 0 && errno == EINTR) {
133 continue;
134 }
135 if (n < 0 && (errno == EAGAIN || errno == EWOULDBLOCK)) {
136 struct pollfd pfd;
137
138 pfd.fd = port->fd;
139 pfd.events = POLLOUT;
140 pfd.revents = 0;
141 if (poll(&pfd, 1u, 100) > 0 && (pfd.revents & POLLOUT) != 0) {
142 continue;
143 }
144 }
145
146 return -1;
147 }
148
149 return 0;
150}

189static int linux_uartbin_app_init(linux_uartbin_app_t *port, const char *path, unsigned baud)
190{
191 uartbin_app_config_t cfg;
192
193 memset(port, 0, sizeof(*port));
194 port->fd = open(path, O_RDWR | O_NOCTTY | O_NONBLOCK);
195 if (port->fd < 0) {
196 return -1;
197 }
198 if (linux_serial_configure(port->fd, baud) != 0) {
199 close(port->fd);
200 port->fd = -1;
201 return -1;
202 }
203
204 memset(&cfg, 0, sizeof(cfg));
205 cfg.write = linux_uartbin_app_write;
206 cfg.on_message = linux_uartbin_app_on_message;
207 cfg.on_delivery = linux_uartbin_app_on_delivery;
208 cfg.on_error = linux_uartbin_app_on_error;
209 cfg.user = port;
210 cfg.rx_payload_buffer = port->rx_payload;
211 cfg.rx_payload_capacity = sizeof(port->rx_payload);
212 cfg.rx_timeout_ms = 50u;
213 cfg.tx_retry_buffer = port->tx_retry_frame;
214 cfg.tx_retry_capacity = sizeof(port->tx_retry_frame);
215 cfg.tx_retry_timeout_ms = UARTBIN_DEFAULT_RETRY_TIMEOUT_MS;
216 cfg.tx_retry_max_retries = UARTBIN_DEFAULT_RETRY_MAX_RETRIES;
217
218 uartbin_app_init(&port->app, &cfg);
219 return 0;
220}

222static int linux_uartbin_app_run(linux_uartbin_app_t *port)
223{
224 for (;;) {
225 struct pollfd pfd;
226 int rc;
227
228 pfd.fd = port->fd;
229 pfd.events = POLLIN | POLLERR | POLLHUP;
230 pfd.revents = 0;
231
232 rc = poll(&pfd, 1u, 10);
233 if (rc < 0 && errno == EINTR) {
234 continue;
235 }
236 if (rc < 0) {
237 return -1;
238 }
239
240 if (rc > 0 && (pfd.revents & POLLIN) != 0) {
241 ssize_t n = read(port->fd, port->rx_block, sizeof(port->rx_block));
242 if (n > 0) {
243 uartbin_app_feed_at(&port->app, port->rx_block, (size_t)n, linux_millis());
244 } else if (n < 0 && errno != EAGAIN && errno != EWOULDBLOCK && errno != EINTR) {
245 uartbin_reset(&port->app.link);
246 }
247 }
248
249 if (rc > 0 && (pfd.revents & (POLLERR | POLLHUP)) != 0) {
250 uartbin_reset(&port->app.link);
251 }
252
253 uartbin_app_poll(&port->app, linux_millis());
254 }
255}

257int main(int argc, char **argv)
258{
259 linux_uartbin_app_t port;
260 const char *path = argc > 1 ? argv[1] : "/dev/ttyUSB0";
261 const uint8_t hello[] = "hello-from-linux";
262
263 if (linux_uartbin_app_init(&port, path, 115200u) != 0) {
264 perror("linux_uartbin_app_init");
265 return 1;
266 }
267
268 if (uartbin_app_send_confirmed(&port.app, MSG_APP_HELLO, 0, hello, (uint16_t)(sizeof(hello) - 1u)) != UARTBIN_OK) {
269 fprintf(stderr, "initial confirmed message could not be queued\n");
270 }
271
272 if (linux_uartbin_app_run(&port) != 0) {
273 perror("linux_uartbin_app_run");
274 close(port.fd);
275 return 1;
276 }
277
278 close(port.fd);
279 return 0;
