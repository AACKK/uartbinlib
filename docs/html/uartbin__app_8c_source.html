44static void uartbin_app_deliver(const uartbin_app_t *app, const uartbin_packet_t *packet)
45{
46 uartbin_app_message_t message;
47
48 if (app->cfg.on_message == 0) {
49 return;
50 }
51
52 message.type = packet->type;
53 message.flags = uartbin_app_public_flags(packet->flags);
54 message.confirmed = ((packet->flags & UARTBIN_APP_FLAG_CONFIRMED) != 0u) ? 1u : 0u;
55 message.seq = packet->seq;
56 message.payload = packet->payload;
57 message.payload_len = packet->payload_len;

69static void uartbin_app_on_packet(const uartbin_packet_t *packet, void *user)
70{
71 uartbin_app_t *app = (uartbin_app_t *)user;
72
73 if (app == 0 || packet == 0) {
74 return;
75 }
76
77 if ((packet->flags & UARTBIN_FLAG_RESPONSE) != 0u) {
78 if ((packet->flags & UARTBIN_APP_FLAG_ACK) != 0u &&
79 app->tx_confirmed_pending != 0u &&
80 packet->seq == app->tx_confirmed_seq) {
81 app->tx_confirmed_pending = 0u;
82 app->tx_confirmed_seq = 0u;
83 if (app->cfg.on_delivery != 0) {
84 app->cfg.on_delivery(packet->seq, app->cfg.user);
85 }
86 }
87 return;
88 }
89
90 if ((packet->flags & UARTBIN_APP_FLAG_CONFIRMED) != 0u) {
91 int duplicate = uartbin_app_is_duplicate_confirmed(app, packet);
92
93 (void)uartbin_send_response(&app->link, packet, packet->type, UARTBIN_APP_FLAG_ACK, 0, 0u);
94 if (duplicate) {
95 return;
96 }
97
98 app->last_rx_confirmed_valid = 1u;
99 app->last_rx_confirmed_type = packet->type;
100 app->last_rx_confirmed_seq = packet->seq;
101 }
102
103 uartbin_app_deliver(app, packet);
104}
